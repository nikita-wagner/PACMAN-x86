buffer times 1024 db 0 ; Buffer to store input text
start:
    ; Initialize buffer pointer
    call clear_screen
    call set_cursor_position
    mov di, buffer

read_input:
    ; Wait for key press
    mov ah, 0
    int 0x16
    cmp al, 13 ; ASCII code for Enter key
    je process_input

    ; Store character in buffer
    mov [di], al
    inc di

    ; Echo character to screen
    mov ah, 0x0e
    int 0x10

    jmp read_input

process_input:
    mov byte [di], 0  ; Null-terminate the buffer
    call clear_screen ; Clear the screen

    call set_cursor_position ; Set cursor position to top-left corner

    mov si, buffer ; Print the input text
    call print_string

    jmp read_input


print_string:
    mov ah, 0x0e
    ret

.repeat:
    lodsb
    cmp al, 0
    je .done
    int 0x10
    jmp .repeat

.done:
    jmp $

clear_screen:
    mov ah, 0x06
    mov al, 0    ; Number of lines to scroll (0 = clear entire screen)
    mov bh, 0x07 ; Attribute (white on black)
    mov cx, 0    ; Start at top-left corner (row 0, column 0)
    mov dx, 0x184f ; End at bottom-right corner (row 24, column 79)
    int 0x10
    ret

set_cursor_position:
    mov ah, 0x02 ; Function to set cursor position
    mov bh, 0x00 ; Page number (0)
    mov dh, 0x00 ; Row (0)
    mov dl, 0x00 ; Column (0)
    int 0x10
    ret

    mov ax, 0x0c00        ; Set AH to 0x0c (BIOS function: write pixel) ; Set AL to 0x01 (pixel color: blue) 
    
    l_b:
    mov cx, 157         ; Set CX to 160 (X-coordinate: 0-319)

    int 0x10            ; Call the BIOS video interrupt

        inc cx
        int 0x10

    mov ax, 0xA000      ; Set AX to 0xA000 (VGA memory segment address)
    mov es, ax          ; Set ES to the VGA memory segment

    mov ax, bx      ;* Set AX to 0x0A0A (pixel color for both bytes)
    mov cx, 320 * 200 / 2 ; Set CX to the number of words to fill (total pixels / 2)
    rep stosw           ; Fill the VGA memory with the pixel color using rep stosw
    ret

draw_pacman:
    mov bx, [pac_x]
    mov si, pac_a      ; Load the address of the array into SI
    mov cx, 13         ; Set the loop counter to the number of elements in the array
    add_0:               ; Load the current element into AL
        lodsb              ; Load byte at [SI] into AL and increment SI
        add al, bl       ; Store the result back into the array
        mov [si-1], al     ; Store AL at [SI-1] (SI was incremented by lodsb) Decrement the loop counter and loop if not zero
        loop add_0

    mov bx, [pac_x]
    mov si, pac_b      ; Load the address of the array into SI
    mov cx, 13         ; Set the loop counter to the number of elements in the array
    add_1:               ; Load the current element into AL
        lodsb              ; Load byte at [SI] into AL and increment SI
        add al, bl       ; Store the result back into the array
        mov [si-1], al     ; Store AL at [SI-1] (SI was incremented by lodsb) Decrement the loop counter and loop if not zero
        loop add_1


    mov ah, 0x02
    int 0x1A
    mov ah, 0x0c  ; BIOS function: write pixel
    mov al, 0x0e  ; Pixel color (yellow)
    xor cx, cx
    xor dx, dx

    mov bx, 0
    S0:
        mov cl, [pac_a + bx]   ; X-coordinate (0-319)
    S1:
        mov dl, bl   ; Y-coordinate (0-199)
        add dl, [pac_y]
        int 0x10
        inc cl
        cmp cl, [pac_b + bx]
        jne S1
        inc bx
        cmp bx, 13
        jne S0

    mov bx, [pac_x]
    mov si, pac_a      ; Load the address of the array into SI
    mov cx, 13         ; Set the loop counter to the number of elements in the array
    add_2:               ; Load the current element into AL
        lodsb              ; Load byte at [SI] into AL and increment SI
        sub al, bl       ; Store the result back into the array
        mov [si-1], al     ; Store AL at [SI-1] (SI was incremented by lodsb) Decrement the loop counter and loop if not zero

horizontal_swap:
    mov bx, pacman_textures   ; Load the offset of pacman_textures

diagonal_swap:
    mov bx, pacman_textures   ; Load the offset of pacman_textures
    mov ax, 0                ; Save the current data segment
    mov ds, ax                ; Set DS back to the original segment
    mov si, bx                ; SI points to the start of the array
    mov di, bx
    add di, 99                ; DI points to the last word of the array (50th word * 2 bytes - 2)
    mov cl, 10
    mov dl, 9
    mov bx, 100
    ds0:
        mov al, [ds:si]           ; Load the first word into AX
        mov ah, [ds:di]           ; Load the last word into DX
        mov [ds:di], al           ; Store the first word at the end
        mov [ds:si], ah           ; Store the last word at the beginning
        sub di, 1
        add si, 10
        loop ds0
    mov ch, 3

        sub si, bx
        inc si
        mov cl, dl

            mov al, [ds:si]           ; Load the first word into AX

            mov [ds:si], ah           ; Store the last word at the beginning

            cmp ch, 0
            jne ds1

texture_swap:
    mov si, pacman_textures  ; SI points to the start of array1
    mov di, pacman_swaptext  ; DI points to the start of array2

        mov ax, [si]    ; Load the value from array1 into AX

        mov [si], ax    ; Store the value from array2 into array1
        add si, 2       ; Move to the next element in array1 (each element is 2 bytes)
        add di, 2       ; Move to the next element in array2
        loop ts0  ; Repeat until CX is 0
    ret

draw_ship:
    call zero_all
    mov al, 0x02
    mov [color], al
    mov si, spaceship_pattern ; Source index pointing to invader_pattern
    call copy_array
    mov di, [spaceship_pos]
    call draw_pattern
    ret
delete_ship:
    call zero_all
    mov al, 0x00
    mov [color], al
    mov di, [spaceship_pos]
    call draw_pattern
    ret

draw_pattern: ; free reg dx
    mov ax, 0xA000 ; m ax
    mov es, ax
    mov bx, 10     ; m bx
    mov ah, 0b00000001
    loop_1:
        mov cx, 8  ; m cx
        loop_0:
            mov al, byte [saved_pattern + bx]
            and al, ah
            rol ah, 1
            add di, 0x140
            cmp al, 0
            jnz draw
            end:
        loop loop_0
        sub di, 0xA00
        inc di
        dec bx
        cmp bx, -1
        jne loop_1
        ret
        draw:
        mov al, [color]
        mov byte [es:di], al
        jmp end
    ret
    
Alien_drawer:
    mov bx, [alien_pos]
    mov dx, [alien_comparator] ; not used in pattern_draw
    mov ax, dx
    rol ax, 1
    mov [alien_comparator], ax
    and dx, [alien_set + bx]
    cmp dx, 0
    jg drawer
    add di, 7+11
    drawer_out:
    mov cl, 1
    add [alien_count], cl
    mov cl, [alien_count]
    cmp cl, 11
    jne Alien_drawer
    add di, (320-198) + 320*10
    mov cl, 1
    add [alien_count + 1], cl
    mov cl, [alien_count + 1]
    mov ch, 0
    mov [alien_count], ch
    mov dx, 1
    mov [alien_comparator], dx
    mov bx, 2
    add [alien_pos], bx
    cmp cl, 5
    jne Alien_drawer
    ret
    drawer:
        call draw_pattern
        add di, 7
        jmp drawer_out

sleep:
    mov dx, 0x1040 ; Low word of microseconds (1000000 microseconds = 1 second) ;mov cx, 0x00   ; High word of microseconds
    mov ah, 0x86   ; Function number for "Wait"
    int 0x15       ; Call the interrupt
    ret

copy_array: ; si array input
    mov di, saved_pattern   ; Destination index pointing to saved_pattern
    mov cx, 11              ; Counter set to 11 bytes to copy
    rep movsb               ; Copy 11 bytes from [SI] to [DI]
    ret                     ; Return from the function                         ; Return from the function

zero_all:
    xor ax, ax
    mov es, ax
    mov ds, ax
    mov [alien_pos], ax
    mov [alien_count], ax
    inc ax
    mov [alien_comparator], ax
    ret

wasd_key:
    mov ah, 0x01
    int 0x16
    jz endw  ; If no key is pressed, jump to no_key_pressed
    call delete_ship
    mov ah, 0x00
    int 0x16           ; Read the key press
    cmp al, 97
    je a_pressed
    cmp al, 100
    je d_pressed
    endw:
    ret
    a_pressed:
        mov ax, 2
        sub [spaceship_pos], ax
        ret
    d_pressed:
        mov ax, 2
        add [spaceship_pos], ax
        ret




draw_bullet:
    mov di, [bullet_pos]
    cmp di, 320
    jbe endb
    sub di, 320
    mov bx, 0xA000
    mov es, bx
    mov bl, 0x0C
    mov byte [es:di], bl
    mov bl, 0x00
    add di, 640*2
    mov byte [es:di], bl
    mov ax, 320
    sub [bullet_pos], ax
    ret
    endb:
    mov ax, [spaceship_pos]
    add ax, 5
    mov [bullet_pos], ax
    ret
